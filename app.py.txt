import streamlit as st
from fpdf import FPDF
import re
import os
import tempfile

# --- CONFIGURAÇÃO DA PÁGINA ---
st.set_page_config(page_title="Gerador de Relatórios", page_icon="📄")

st.title("📄 Gerador de Relatórios Faturamento")
st.write("Faça upload dos arquivos `.TXT` (Layout FATUCOPA) e baixe os PDFs formatados.")

# --- CLASSE PDF (Igual a anterior) ---
class PDF(FPDF):
    def header(self):
        self.set_font('Arial', 'B', 14)
        self.cell(0, 10, 'RELATÓRIO DE DADOS PROCESSADOS', 0, 1, 'C')
        self.line(10, 20, 200, 20)
        self.ln(10)

    def chapter_title(self, label):
        self.set_font('Arial', 'B', 12)
        self.set_fill_color(240, 240, 240)
        self.cell(0, 8, label, 0, 1, 'L', 1)
        self.ln(2)

    def chapter_body(self, body):
        self.set_font('Arial', '', 11)
        self.multi_cell(0, 6, body)
        self.ln(3)

# --- FUNÇÃO DE EXTRAÇÃO (REGEX) ---
def extrair_dados(texto_completo):
    dados = {
        "nome": "NÃO ENCONTRADO",
        "valor": "0,00",
        "competencia": "N/D",
        "nascimento": "N/D",
        "codigo": "N/D"
    }

    # 1. Datas
    datas = re.findall(r'\d{2}/\d{2,4}', texto_completo)
    for d in datas:
        if len(d) == 7: # ex: 12/2025
            dados['competencia'] = d
        elif len(d) == 10 and '2026' not in d: 
            dados['nascimento'] = d

    # 2. Nome
    match_nome = re.search(r'[A-Z\s]{10,}', texto_completo)
    if match_nome:
        dados['nome'] = match_nome.group(0).strip()

    # 3. Valor
    match_valor = re.search(r'0{5,}(\d+)', texto_completo)
    if match_valor:
        valor_bruto = int(match_valor.group(1))
        dados['valor'] = f"R$ {valor_bruto/100:,.2f}".replace('.', ',')

    # 4. Código
    match_cod = re.search(r'[0-9]X[0-9A-Z]{3}', texto_completo)
    if match_cod:
        dados['codigo'] = match_cod.group(0)

    return dados

# --- FUNÇÃO PARA GERAR PDF EM MEMÓRIA ---
def gerar_pdf_bytes(nome_arquivo, dados):
    pdf = PDF()
    pdf.add_page()
    
    pdf.chapter_title(f'Arquivo Original: {nome_arquivo}')
    pdf.chapter_body(f"Extração automática via Web App.")

    pdf.chapter_title('Dados do Cliente')
    pdf.chapter_body(f"Nome: {dados['nome']}\nData de Nascimento: {dados['nascimento']}\nCódigo: {dados['codigo']}")

    pdf.chapter_title('Dados Financeiros')
    pdf.chapter_body(f"Competência: {dados['competencia']}\nValor Líquido: {dados['valor']}")
    
    # Salva em um arquivo temporário para garantir compatibilidade
    with tempfile.NamedTemporaryFile(delete=False, suffix=".pdf") as tmp_file:
        pdf.output(tmp_file.name)
        return tmp_file.name

# --- INTERFACE DO USUÁRIO ---
uploaded_files = st.file_uploader("Arraste seus arquivos aqui", type=["txt"], accept_multiple_files=True)

if uploaded_files:
    st.success(f"{len(uploaded_files)} arquivo(s) carregado(s)!")
    
    for uploaded_file in uploaded_files:
        # Lê o arquivo
        string_data = uploaded_file.getvalue().decode("utf-8", errors="ignore")
        
        # Processa
        dados = extrair_dados(string_data)
        
        # Mostra prévia na tela
        with st.expander(f"Verificar: {dados['nome']} - {dados['valor']}"):
            st.write(dados)
            
            # Gera o PDF
            caminho_pdf_temp = gerar_pdf_bytes(uploaded_file.name, dados)
            
            # Botão de Download
            with open(caminho_pdf_temp, "rb") as f:
                btn = st.download_button(
                    label=f"⬇️ Baixar PDF de {dados['nome']}",
                    data=f,
                    file_name=f"Relatorio_{dados['nome'].replace(' ', '_')}.pdf",
                    mime="application/pdf"
                )
